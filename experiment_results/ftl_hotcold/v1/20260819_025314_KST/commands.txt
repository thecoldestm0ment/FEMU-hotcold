Guest destructive-write safety gate
-----------------------------------
date --iso-8601=seconds
lsblk -b -o NAME,SIZE,TYPE,FSTYPE,MOUNTPOINT
sudo -n blockdev --getsize64 /dev/nvme0n1
findmnt /dev/nvme0n1; echo findmnt_rc=$?
test "$(sudo -n blockdev --getsize64 /dev/nvme0n1)" = 6442450944
test ! -e /dev/nvme0n1p1
fio --version
nvme version

Guest preconditioning
---------------------
sudo -n fio --name=fill --filename=/dev/nvme0n1 --direct=1 --ioengine=libaio --rw=write --bs=1M --iodepth=32 --numjobs=1 --size=6G --group_reporting=1
echo fill_rc=$?

Marker, precondition stats, and measurement reset
-------------------------------------------------
printf V1GC | sudo -n dd of=/dev/nvme0n1 bs=4 count=1 conv=notrunc status=none
printf V1GC | sudo -n dd of=/dev/nvme0n1 bs=4 count=1 conv=notrunc status=none
sync
sudo -n nvme admin-passthru /dev/nvme0 --opcode=0xef --cdw10=5

Guest measurement (T substituted in job and filenames)
------------------------------------------------------
nohup sh -c 'date --iso-8601=seconds; sudo -n fio --name=v1-t<T> --filename=/dev/nvme0n1 --offset=1G --size=4G --direct=1 --ioengine=libaio --rw=randwrite --bs=16k --iodepth=128 --numjobs=32 --time_based=1 --runtime=300 --random_distribution=zipf:0.99 --randrepeat=1 --randseed=20260819 --group_reporting=1; rc=$?; date --iso-8601=seconds; echo measurement_rc=$rc; echo $rc > /tmp/v1_t<T>_measure.rc' > /tmp/v1_t<T>_measure.out 2>&1 < /dev/null &
